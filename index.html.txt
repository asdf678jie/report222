<script>
    // 导航切换
    const navItems = document.querySelectorAll('.nav-item');
    const pages = document.querySelectorAll('.page');
    function switchPage(pageId){
        navItems.forEach(i=>i.classList.remove('active'));
        pages.forEach(p=>p.classList.remove('active'));
        document.querySelector(`[data-page="${pageId}"]`).classList.add('active');
        document.getElementById(pageId).classList.add('active');
    }
    navItems.forEach(item=>{
        item.onclick = ()=>switchPage(item.dataset.page);
    })

    // 文件上传模拟
    const uploadArea = document.getElementById('uploadArea');
    const fileInput = document.getElementById('fileInput');
    uploadArea.onclick = ()=>fileInput.click();
    fileInput.onchange = function(){
        showMsg('文件上传成功，正在进行OCR文本抽取与数据识别');
    }

    // 计税模拟
    function calcTax(){
        document.getElementById('taxResult').style.display='block';
        showMsg('计税完成，自动生成申报表，支持人工复核调整');
    }

    // 消息弹窗
    function showMsg(text){
        const m = document.getElementById('msg');
        m.innerText = text;
        m.style.display='block';
        setTimeout(()=>m.style.display='none',2200);
    }

    // 智能问答
    function sendQa(){
        const input = document.getElementById('qaInput');
        const box = document.getElementById('chatBox');
        const text = input.value.trim();
        if(!text) return;
        box.innerHTML += `<div class="chat-msg chat-user">用户：${text}</div>`;
        // 固定财税问答回复（演示）
        let ans = "根据税收知识图谱推理：小高有限公司为西安小规模商贸企业，可以享受小规模增值税减免、小微企业所得税优惠、陕西六税两费政策；请留意留存备查资料，按时申报。";
        box.innerHTML += `<div class="chat-msg chat-robot">财税智能体：${ans}</div>`;
        box.scrollTop = box.scrollHeight;
        input.value = "";
    }
</script>
